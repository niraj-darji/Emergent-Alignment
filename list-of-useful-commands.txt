curl -LsSf https://astral.sh/uv/install.sh | sh
source ~/.bashrc

# Create a Python 3.11 environment and install dependencies
uv venv --python 3.11
source .venv/bin/activate
uv pip install -r requirements.txt


# Models like Llama 3.1 and Gemma are gated models. 
# You must accept their license on Hugging Face and generate a User Access Token (Read)
# once you accept their license & you see 
# Request Status "ACCEPTED" at "https://huggingface.co/settings/gated-repos", you can loging HF cli and download the weights.

huggingface-cli login

huggingface-cli download google/gemma-3-12b-it \
  --token your_hf_token \
  --local-dir /your/local/dir 

hf download Qwen/Qwen3-8B \
  --token your_hf_token \
  --local-dir /your/local/dir 

hf download meta-llama/Llama-3.1-8B-Instruct \
  --token your_hf_token \
  --local-dir /your/local/dir
